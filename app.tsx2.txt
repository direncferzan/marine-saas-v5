// --- ANA COMPONENT ---
export default function App() {
  const [restaurants, setRestaurants] = useState<Restaurant[]>(() => {
    const saved = localStorage.getItem("marine_res");
    return saved ? JSON.parse(saved) : INITIAL_RESTAURANTS;
  });

  const [users, setUsers] = useState<AppUser[]>(() => {
    const saved = localStorage.getItem("marine_users");
    return saved
      ? JSON.parse(saved)
      : [
          {
            id: "admin-1",
            name: "Marine Admin",
            email: "admin@marine.com",
            pass: "admin123",
            role: "Admin",
          },
        ];
  });

  const [activeUser, setActiveUser] = useState<AppUser | null>(null);
  const [cart, setCart] = useState<any[]>([]);
  const [orders, setOrders] = useState<Order[]>([]);

  useEffect(() => {
    localStorage.setItem("marine_res", JSON.stringify(restaurants));
    localStorage.setItem("marine_users", JSON.stringify(users));
  }, [restaurants, users]);

  // --- ADMIN WATCHDOG MODÜLLERİ ---
  const watchdogCreateRestaurant = (name: string, email: string) => {
    const resId = "res-" + Date.now();
    const tempPass = "PASS" + Math.floor(Math.random() * 9000);

    const newRes: Restaurant = {
      id: resId,
      name,
      ownerEmail: email,
      tempPass,
      logo: name.substring(0, 2).toUpperCase(),
      cover: "https://images.pexels.com/photos/262978/pexels-photo-262978.jpeg",
      rating: 5.0,
      eta: "30-40 dk",
      minOrder: 150,
      deliveryFee: 20,
      isOpen: true,
      menu: [],
    };

    const newUser: AppUser = {
      id: "u-" + Date.now(),
      name: name + " Yöneticisi",
      email,
      pass: tempPass,
      role: "RestaurantOwner",
      restaurantId: resId,
    };

    setRestaurants([...restaurants, newRes]);
    setUsers([...users, newUser]);
  };

  // --- UI RENDERLARI ---
  return (
    <BrowserRouter>
      <div className="min-h-screen bg-[#f7f7f7]">
        {/* TOPBAR */}
        <nav className="topbar">
          <div className="app-shell flex justify-between items-center w-full">
            <Link to="/" className="flex items-center gap-2 no-underline">
              <div className="bg-[#fa0050] text-white p-2 rounded-xl font-black text-xl">

              </div>

                    </p>

                  </div>

                </div>
              ) : (
                <Link to="/login" className="btn-ys no-underline py-2 px-6">
                  Giriş Yap
                </Link>
              )}
            </div>
          </div>
        </nav>

          <Route path="/" element={<Home restaurants={restaurants} />} />

            path="/login"

function RestaurantDetail({ restaurants, cart, setCart }: any) {
  const { id } = useParams();
  const navigate = useNavigate();
  const res = restaurants.find((r: any) => r.id === id);
  const [selectedProduct, setSelectedProduct] = useState<MenuItem | null>(null);
  const [activeMods, setActiveMods] = useState<Modifier[]>([]);

  if (!res) return <div>Restoran bulunamadı.</div>;

  const handleAddToCart = () => {
    if (!selectedProduct) return;
    const modTotal = activeMods.reduce((a, b) => a + b.price, 0);
    const newItem = {
      ...selectedProduct,
      selectedMods: [...activeMods],
      totalPrice: selectedProduct.price + modTotal,
    };
    setCart([...cart, newItem]);
    setSelectedProduct(null);
    setActiveMods([]);
  };

  return (
    <div className="app-shell py-8 flex flex-col lg:flex-row gap-12">
      <div className="flex-1">
        <div className="relative h-80 rounded-[40px] overflow-hidden mb-12 shadow-2xl">
          <img src={res.cover} className="w-full h-full object-cover" />
          <div className="absolute inset-0 bg-gradient-to-t from-black/80 flex flex-col justify-end p-10 text-white">
            <h1 className="text-5xl font-black">{res.name}</h1>
            <p className="opacity-70 mt-2 font-medium">
              En sevilen Marine lezzetleri
            </p>
          </div>
        </div>

        <div className="space-y-6">
          <h2 className="text-2xl font-black mb-8 border-b-4 border-pink-100 inline-block">
            Menü
          </h2>
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {res.menu.map((item: MenuItem) => (
              <div
                key={item.id}
                onClick={() => setSelectedProduct(item)}
                className="bg-white p-4 rounded-2xl flex justify-between gap-4 cursor-pointer hover:border-pink-500 border-2 border-transparent shadow-sm"
              >
                <div className="flex-1">
                  <h4 className="font-bold text-lg">{item.name}</h4>
                  <p className="text-xs text-gray-400 mt-1 line-clamp-2">
                    {item.description}
                  </p>
                  <p className="mt-3 font-black text-pink-600">
                    {item.price} TL
                  </p>
                </div>
                <img
                  src={item.image}
                  className="w-24 h-24 rounded-xl object-cover"
                />
              </div>
            ))}
          </div>
        </div>
      </div>

      {/* PRODUCT MODAL */}
      {selectedProduct && (
        <div className="fixed inset-0 bg-black/70 backdrop-blur-md z-[2000] flex items-center justify-center p-4">
          <div className="bg-white w-full max-w-xl rounded-[40px] overflow-hidden animate-in zoom-in-95">
            <div className="relative h-56">
              <img
                src={selectedProduct.image}
                className="w-full h-full object-cover"
              />
              <button
                onClick={() => setSelectedProduct(null)}
                className="absolute top-6 right-6 bg-white/80 p-2 rounded-full"
              >
                <X />
              </button>
            </div>
            <div className="p-8">
              <h2 className="text-3xl font-black">{selectedProduct.name}</h2>
              <p className="text-gray-500 mt-2">
                {selectedProduct.description}
              </p>

              <div className="mt-8 space-y-3">
                <h4 className="font-black text-sm uppercase tracking-widest text-gray-400">
                  Malzeme Seçimi (Ekle/Çıkar)
                </h4>
                {selectedProduct.modifiers.map((mod) => (
                  <div
                    key={mod.id}
                    onClick={() => {
                      if (activeMods.find((m) => m.id === mod.id)) {
                        setActiveMods(
                          activeMods.filter((m) => m.id !== mod.id)
                        );
                      } else {
                        setActiveMods([...activeMods, mod]);
                      }
                    }}
                    className={`modifier-badge cursor-pointer ${
                      activeMods.find((m) => m.id === mod.id) ? "active" : ""
                    }`}
                  >
                    <span>{mod.name}</span>
                    <span className="font-black">
                      {mod.price > 0 ? `+${mod.price} TL` : "Ücretsiz"}
                    </span>
                  </div>
                ))}
              </div>

              <div className="mt-10 border-t pt-6 flex justify-between items-center">
                <div>
                  <p className="text-xs font-bold text-gray-400">
                    TOPLAM FİYAT
                  </p>
                  <p className="text-3xl font-black text-gray-900">
                    {selectedProduct.price +
                      activeMods.reduce((a, b) => a + b.price, 0)}{" "}
                    TL
                  </p>
                </div>
                <button
                  onClick={handleAddToCart}
                  className="btn-ys h-16 px-12 text-lg"
                >
                  Sepete Ekle
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

function Login({ users, onLogin }: any) {
  const [email, setEmail] = useState("");
  const [pass, setPass] = useState("");
  const navigate = useNavigate();

  const handleLogin = (e: React.FormEvent) => {
    e.preventDefault();
    const user = users.find((u: any) => u.email === email && u.pass === pass);
    if (user) {
      onLogin(user);
      if (user.role === "Admin") navigate("/admin");
      else if (user.role === "RestaurantOwner") navigate("/owner");
      else navigate("/");
    } else {
      alert("Hatalı bilgiler!");
    }
  };

  return (
    <div className="flex items-center justify-center min-h-[80vh] p-4">
      <div className="bg-white p-10 rounded-[40px] shadow-2xl w-full max-w-md">
        <h2 className="text-3xl font-black mb-8">Marine Giriş</h2>
        <form onSubmit={handleLogin} className="space-y-4">
          <input
            className="w-full p-4 bg-gray-50 border-none rounded-2xl outline-none focus:ring-2 ring-pink-500"
            placeholder="E-posta"
            onChange={(e) => setEmail(e.target.value)}
          />
          <input
            type="password"
            className="w-full p-4 bg-gray-50 border-none rounded-2xl outline-none focus:ring-2 ring-pink-500"
            placeholder="Şifre"
            onChange={(e) => setPass(e.target.value)}
          />
          <button className="btn-ys w-full py-5 text-lg mt-4">Giriş Yap</button>
        </form>
      </div>
    </div>
  );
}
